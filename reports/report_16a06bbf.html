📊 Approach & Query

Query Code


WITH sales_data AS (
    SELECT 
        s.store_id,
        s.product_id,
        -- Normalize category names to handle inconsistent casing and variations
        CASE 
            WHEN UPPER(s.category) IN ('ELECTRONICS', 'ELECTRONIC', 'ELEC.') THEN 'Electronics'
            WHEN UPPER(s.category) IN ('BEAUTY', 'COSMETICS') THEN 'Beauty'
            WHEN UPPER(s.category) IN ('SPORTS', 'SPORT', 'SPORTING GOODS') THEN 'Sports'
            WHEN UPPER(s.category) IN ('APPAREL', 'CLOTHING') THEN 'Apparel'
            WHEN UPPER(s.category) IN ('HOME & KITCHEN', 'HOME') THEN 'Home & Kitchen'
            ELSE 'Other'
        END as category,
        s.total_amount as revenue,
        st.region
    FROM read_csv_auto('C:\Users\amarkkumar\Downloads\Amar_usecase\amar_case\data\landing\sales_data 1 4.csv') s
    JOIN read_csv_auto('C:\Users\amarkkumar\Downloads\Amar_usecase\amar_case\data\landing\stores 1 4.csv') st
    ON s.store_id = st.store_id
),
category_region_revenue AS (
    SELECT 
        region,
        category,
        SUM(revenue) as total_revenue,
        COUNT(*) as transaction_count
    FROM sales_data
    GROUP BY region, category
),
regional_totals AS (
    SELECT 
        region,
        SUM(total_revenue) as region_total_revenue
    FROM category_region_revenue
    GROUP BY region
),
ranked_categories AS (
    SELECT 
        crr.region,
        crr.category,
        crr.total_revenue,
        crr.transaction_count,
        rt.region_total_revenue,
        ROUND(100.0 * crr.total_revenue / rt.region_total_revenue, 2) as pct_of_regional_revenue,
        ROW_NUMBER() OVER (PARTITION BY crr.region ORDER BY crr.total_revenue DESC) as rank_in_region,
        ROW_NUMBER() OVER (ORDER BY crr.total_revenue DESC) as overall_rank
    FROM category_region_revenue crr
    JOIN regional_totals rt ON crr.region = rt.region
),
top_categories_per_region AS (
    SELECT 
        region,
        category as top_category,
        ROUND(total_revenue, 2) as top_category_revenue,
        transaction_count as top_category_transactions,
        ROUND(region_total_revenue, 2) as region_total_revenue,
        pct_of_regional_revenue as top_category_pct_share
    FROM ranked_categories
    WHERE rank_in_region = 1
)
SELECT 
    rc.region,
    rc.category,
    ROUND(rc.total_revenue, 2) as total_revenue,
    rc.transaction_count,
    ROUND(rc.region_total_revenue, 2) as region_total_revenue,
    rc.pct_of_regional_revenue,
    rc.rank_in_region,
    CASE WHEN rc.rank_in_region = 1 THEN 'TOP PERFORMER' ELSE '' END as top_performer_flag
FROM ranked_categories rc
ORDER BY rc.region, rc.rank_in_region

Query Description

Joined sales transaction data with store location data, normalized inconsistent category naming (Electronics/ELECTRONICS/Electronic/Elec. → Electronics), calculated total revenue by category within each region, ranked categories by revenue performance per region, and computed percentage share of regional revenue for comparative analysis.
